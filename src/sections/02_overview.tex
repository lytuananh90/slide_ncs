\begin{frame}
    \frametitle{Giới thiệu tổng quan}
    \SubTitle{Bối cảnh và động cơ}
    \begin{BulletTight}
        \item Kiểm thử là giai đoạn then chốt đảm bảo chất lượng phần mềm.
        \item Chi phí sửa lỗi sau phát hành cao gấp 10–100 lần phát hiện sớm.%
        \src{1}{Boehm, B. W. (1981). \textit{Software engineering economics}. Englewood Cliffs, NJ: Prentice-Hall.}
        \item Kiểm thử thủ công tốn kém, phụ thuộc con người.
        \item Phần mềm ngày càng phức tạp, thay đổi nhanh, khó bao phủ kiểm thử toàn diện.
        \item Tự động hóa kiểm thử trở thành xu hướng tất yếu.
        \item[\raise0.2ex\hbox{\color{KHTN_BLUE!85}\scriptsize\faArrowRight}]
        Cần hướng đến phương pháp kiểm thử \textit{thông minh, thích ứng cao và có khả năng hiểu ngữ cảnh.}
    \end{BulletTight}
\end{frame}

% SLIDE 5: GIAI ĐOẠN TRƯỚC LLM

\begin{frame}
    \frametitle{Giới thiệu tổng quan}
    \SubTitle{Giai đoạn trước khi LLM bùng nổ}

    \begin{BulletTight}
        \item Kiểm thử chủ yếu dựa vào kịch bản thủ công.
        \item Một số mô hình học tăng cường sinh test tự động.%
        \src{2}{Nguyen, A., Le, B., \& Nguyen, V. (2019). Prioritizing automated UI tests using reinforcement learning. \textit{Proceedings of PROMISE}, 56–65.}
        \item Các mô hình này khó thích ứng khi phần mềm thay đổi nhanh.
        \item Thiếu khả năng hiểu ngữ nghĩa và bối cảnh kiểm thử.
        \item[\raise0.2ex\hbox{\color{KHTN_BLUE!85}\scriptsize\faArrowRight}]
        Giới hạn về tự động hóa và khả năng hiểu ngữ cảnh.
    \end{BulletTight}
\end{frame}

\begin{frame}
    \frametitle{Giới thiệu tổng quan}
    \SubTitle {Các nghiên cứu tiêu biểu ứng dụng LLM - đơn tác tử}

    \begin{BulletTight}
        \item Dễ triển khai, hiệu quả cho các tác vụ riêng lẻ.
        \item Có khả năng hiểu ngữ cảnh và sinh test hợp lý.
        \item[\raise0.2ex\hbox{\color{KHTN_BLUE!85}\scriptsize\faArrowRight}]
        LLM tạo bước tiến lớn trong tự động hóa kiểm thử.
        \item Một tác tử xử lý toàn bộ quy trình, thiếu phối hợp.%
        \src{5}{Fan, Y. et al. (2024). Can cooperative multi-agent reinforcement learning boost automatic web testing? \textit{ASE}, 14–26.}
        \item Dễ lặp hành động, bỏ sót lỗi hoặc “ảo giác” mục tiêu.%
        \item[\raise0.2ex\hbox{\color{KHTN_BLUE!85}\scriptsize\faArrowRight}]
        Thiếu phối hợp — chưa phù hợp cho kiểm thử quy mô lớn, đa nhiệm.
    \end{BulletTight}
\end{frame}
