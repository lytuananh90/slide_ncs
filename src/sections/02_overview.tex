\begin{frame}
    \frametitle{Các hướng nghiên cứu kiểm thử tự động}

    \begin{BulletTight}

        \item \textbf{Kiểm thử giao diện (UI Testing)}
        \begin{itemize}
            \item RL điều hướng UI và tối ưu hành vi test.%
            \src{1}{Nguyen, A. et al. (2019). RL for UI test prioritization. PROMISE.}
            \item LLM khám phá UI theo hành vi người dùng.%
            \src{2}{Ran, D. et al. (2024). Guardian. ISSTA.}
            \item Multi-agent tăng bao phủ UI.%
            \src{3}{Feng, S. et al. (2025). Cooperative agents for web testing. ASE.}
        \end{itemize}

        \vspace{0.4em}
        \item \textbf{Kiểm thử API}
        \begin{itemize}
            \item LLM khai phá ràng buộc API \& sinh test.%
            \src{4}{Huynh, H., Le, Q. T. et al. (2024). KAT for API testing. ASE.}
        \end{itemize}

        \vspace{0.4em}
        \item \textbf{Sửa lỗi tự động (Program Repair)}
        \begin{itemize}
            \item LLM hiểu mã \& sửa lỗi logic.%
            \src{5}{Jin, M. et al. (2023). InferFix. ESEC/FSE.}
        \end{itemize}

    \end{BulletTight}

\end{frame}

\begin{frame}
    \frametitle{Kiểm thử tự động}


    \begin{BulletTight}
        \item Tăng tốc độ kiểm thử, rút ngắn chu kỳ phát triển.
        \item Giảm chi phí nhân lực, hạn chế lỗi do con người.%
        \src{1}{Boehm, B. (1981). Software engineering economics.}
        \item Đảm bảo tính lặp lại, ổn định và bao phủ cao hơn kiểm thử thủ công.
        \item Phù hợp DevOps/CI-CD, hỗ trợ kiểm thử liên tục.
        \item Mở rộng sang sử dụng RL/LLM để sinh test thông minh.%
        \src{2}{Nguyen, A. et al. (2019). RL for UI testing. PROMISE.}
    \end{BulletTight}

\end{frame}
