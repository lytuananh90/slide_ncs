\begin{frame}
    \frametitle{Giới thiệu tổng quan}
    \SubTitle{Các hướng nghiên cứu kiểm thử tự động}

    \begin{BulletTight}

        \item \textbf{Kiểm thử giao diện (UI Testing)}
        \begin{itemize}
            \item RL điều hướng UI \& ưu tiên test — Nguyen et al., 2019 (PROMISE)
            \item LLM khám phá UI theo hành vi — Ran et al., 2024 (ISSTA)
            \item Đa tác tử RL tăng bao phủ UI — Fan et al., 2024 (ASE)
        \end{itemize}

        \vspace{0.3em}
        \item \textbf{Kiểm thử API}
        \begin{itemize}
            \item LLM khai phá ràng buộc API \& sinh test — Huynh et al., 2024 (ASE EA)
        \end{itemize}

        \vspace{0.3em}
        \item \textbf{Sửa lỗi tự động (Program Repair)}
        \begin{itemize}
            \item LLM hiểu mã \& sửa lỗi logic — Jin et al., 2023 (ESEC/FSE)
        \end{itemize}

    \end{BulletTight}
\end{frame}
