
\begin{frame}
    \frametitle{Kết luận}

    \begin{BulletTight}

        \item Đề tài hướng tới hệ kiểm thử tự động dựa trên kiến trúc LLM-đa tác tử, khắc phục một số hạn chế của những mô hình kiểm thử hiện tại và tăng hiệu quả kiểm thử.

        \item Các mục tiêu chính:
        \begin{itemize}
            \item Xây dựng hệ đa tác tử tích hợp LLM
            \item Cải thiện cơ chế giao tiếp, phối hợp giữa các tác tử
            \item Tối ưu cơ chế HITL
            \item Nghiên cứu sâu một tác tử tiêu biểu để đảm bảo chiều sâu đề tài
            \item Xây dựng framework mã nguồn mở để cộng đồng mở rộng
            \item Có những đóng góp khoa học và thực tiễn rõ ràng
        \end{itemize}

    \end{BulletTight}

\end{frame}
