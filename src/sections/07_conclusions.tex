\begin{frame}
    \frametitle{Kết luận}

    \begin{BulletTight}
        \item Đề tài hướng đến hệ kiểm thử tự động thông minh dựa trên LLM và hệ đa tác tử.
        \item Giải quyết các hạn chế của hệ đơn tác tử: thiếu phối hợp, dễ ảo giác, khó mở rộng.
        \item Đề xuất cơ chế cộng tác và phản hồi con người (HITL) giúp tăng độ tin cậy.
        \item Xây dựng nền tảng có khả năng tái sử dụng và mở rộng cho nghiên cứu tương lai.
        \item[\raise0.2ex\hbox{\color{KHTN_BLUE!85}\scriptsize\faArrowRight}]
        \textbf{Định hướng tiếp theo:} triển khai thực nghiệm sâu hơn và mở rộng sang kiểm thử toàn hệ thống.
    \end{BulletTight}
\end{frame}
