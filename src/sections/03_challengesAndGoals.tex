\begin{frame}
    \frametitle{Thách thức}
    \SubTitle{Những rào cản trong nghiên cứu hệ đa tác tử}

    \begin{BulletTight}
        \item \textbf{Ảo giác (Hallucination):}
        Tác tử sinh phản hồi sai lệch, gây lỗi kiểm thử.%
        \src{1}{Fan, Y. et al. (2024). \textit{Cooperative MA-RL for Web Testing}. ASE.}

        \item \textbf{Giao tiếp:}
        Dễ mất ngữ cảnh, lan truyền lỗi, phối hợp chưa hiệu quả.%
        \src{2}{He, J. et al. (2025). \textit{LLM-Based Multi-Agent Systems}. TOSEM.}

        \item \textbf{HITL:}
        Phản hồi con người thiên kiến, giám sát tốn chi phí.%
        \src{3}{Xu, Q. et al. (2025). \textit{Hallucination to Consensus}. arXiv.}

        \item \textbf{Hiệu năng:}
        Nhiều LLM song song, tốn tài nguyên và tăng độ trễ.%
        \src{4}{Hong, S. et al. (2024). \textit{MetaGPT}. ICLR.}

        \item[\raise0.2ex\hbox{\color{KHTN_BLUE!85}\scriptsize\faArrowRight}]
        \textbf{→ Cần hệ đa tác tử ổn định, giao tiếp hiệu quả và có HITL.}
    \end{BulletTight}
\end{frame}

\begin{frame}
  \frametitle{Mục tiêu nghiên cứu}
  \SubTitle{Hệ kiểm thử thông minh và đáng tin cậy}
  \begin{BulletTight}
    \item \textbf{Xây dựng hệ đa tác tử:}
      \begin{BulletTight}
        \item Xác định vai trò chuyên biệt (Planner, Executor, ...).
        \item Thiết kế cơ chế phối hợp linh hoạt, hiệu quả.
      \end{BulletTight}
    \item \textbf{Nâng cao giao tiếp và độ tin cậy:}
      \begin{BulletTight}
        \item Giảm nhầm lẫn ngữ cảnh và mâu thuẫn thông tin.
        \item Hạn chế "ảo giác" (hallucination) thông qua kiểm tra chéo.
      \end{BulletTight}
    \item \textbf{Tối ưu hóa Human-in-the-Loop (HITL):}
      \begin{BulletTight}
        \item Xác định thời điểm can thiệp của con người.
        \item Tận dụng tri thức con người, giảm thiên kiến.
      \end{BulletTight}
    \item \textbf{Tối ưu hiệu năng và chi phí:}
      \begin{BulletTight}
        \item Giảm số lượng lời gọi LLM, định tuyến mô hình thông minh.
      \end{BulletTight}
  \end{BulletTight}
\end{frame}

% ======================================================
\begin{frame}
    \frametitle{Đối tượng và phạm vi nghiên cứu}

    \begin{BulletTight}
        \item \textbf{Đối tượng nghiên cứu:}
        \begin{itemize}
            \setlength{\itemsep}{0.3em}
            \item Hệ thống đa tác tử (Multi-Agent System) tích hợp mô hình ngôn ngữ lớn (LLM) cho kiểm thử phần mềm tự động.
            \item Các tác tử chính: Planner, Executor, Evaluator, Coordinator và HITL Agent.
            \item Cơ chế phối hợp, giao tiếp và phản hồi giữa các tác tử trong quá trình kiểm thử.
        \end{itemize}

        \vspace{0.7em}
        \item \textbf{Phạm vi nghiên cứu:}
        \begin{itemize}
            \setlength{\itemsep}{0.3em}
            \item Ứng dụng cho kiểm thử phần mềm web và di động.
            \item Tập trung phát triển và phân tích sâu \textbf{một tác tử tiêu biểu} nhằm tối ưu hiệu năng và khả năng phối hợp.
            \item Không mở rộng sang kiểm thử hiệu năng hay bảo mật.
        \end{itemize}

        \vspace{0.6em}
        \item[\raise0.2ex\hbox{\color{KHTN_BLUE!85}\scriptsize\faArrowRight}]
        \textbf{Tóm lược:} Nghiên cứu tập trung vào thiết kế cơ chế phối hợp đa tác tử và cải thiện độ tin cậy của kiểm thử tự động.
    \end{BulletTight}
\end{frame}

\begin{frame}
  \frametitle{Phương pháp nghiên cứu}
  \SubTitle{Thiết kế thực nghiệm}
  \begin{BulletTight}
    \item \textbf{Khảo sát và phân tích:}
      \begin{BulletTight}
        \item Tổng hợp công trình, xác định khoảng trống nghiên cứu.
      \end{BulletTight}
    \item \textbf{Thiết kế mô hình:}
      \begin{BulletTight}
        \item Xây dựng kiến trúc đa tác tử và cơ chế giao tiếp.
      \end{BulletTight}
    \item \textbf{Phát triển nguyên mẫu:}
      \begin{BulletTight}
        \item Triển khai các tác tử chính, tích hợp cơ chế HITL.
      \end{BulletTight}
    \item \textbf{Thực nghiệm và đánh giá:}
      \begin{BulletTight}
        \item Đo lường hiệu năng, độ chính xác, so sánh với baseline.
      \end{BulletTight}
    \item \textbf{Hoàn thiện và tổng kết:}
      \begin{BulletTight}
        \item Phân tích kết quả, tối ưu mô hình, hoàn thiện luận án.
      \end{BulletTight}
  \end{BulletTight}
\end{frame}
