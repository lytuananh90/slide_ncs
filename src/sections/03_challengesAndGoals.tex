\begin{frame}
    \frametitle{Mục tiêu nghiên cứu}
    \SubTitle{Xây dựng hệ đa tác tử tích hợp LLM cho kiểm thử tự động}

    \begin{BulletTight}
        \item \textbf{Tích hợp, phối hợp nhiều tác tử có nhiệm vụ cụ thể}
        \begin{itemize}
            \item Khám phá tốt hơn trên không gian kiểm thử lớn
            \item Chuyên môn hóa tác tử
            \item Tối ưu cơ chế phối hợp giữa các tác tử $\rightarrow$ Xử lý những hệ thống lớn và luồng kiểm thử phức tạp
            \item Dễ dàng tích hợp tác tử mới, khả năng tái sử dụng cao
        \end{itemize}

        \vspace{0.4em}
        \item \textbf{Tích hợp và tối ưu cơ chế HITL}
        \begin{itemize}
            \item Hỗ trợ xác định các oracle khó
            \item Giám sát \& điều chỉnh hành vi tác tử
            \item Học từ phản hồi con người
        \end{itemize}
    \end{BulletTight}
\end{frame}
