% ======================================================
\begin{frame}
    \frametitle{Thách thức của đề tài}
    \begin{BulletTight}
        \item \textbf{Ảo giác (Hallucination):}
        Tác tử sinh phản hồi sai lệch%

        \item \textbf{Giao tiếp:}
        Dễ mất ngữ cảnh, lan truyền lỗi%

        \item \textbf{HITL:}
        Phản hồi con người thiên kiến, giám sát tốn chi phí.%

        \item \textbf{Hiệu năng:}
        Nhiều LLM song song, tốn tài nguyên và tăng độ trễ.%

    \end{BulletTight}
\end{frame}

% ======================================================
% SLIDE 10: MỤC TIÊU NGHIÊN CỨU
% ======================================================
\begin{frame}
    \frametitle{Mục tiêu nghiên cứu}

    \begin{BulletTight}
        \item Xây dựng hệ đa tác tử tích hợp LLM cho kiểm thử tự động.
        \item Xác định vai trò và cơ chế phối hợp linh hoạt.
        \item Nghiên cứu sâu một tác tử tiêu biểu để tối ưu hiệu năng.
        \item Cải thiện giao tiếp và giảm lỗi trong kiểm thử phối hợp.
        \item Tối ưu HITL để tận dụng tri thức con người.
        \item Giảm chi phí, tăng độ chính xác và khả năng mở rộng.

    \end{BulletTight}
\end{frame}

% ======================================================
% SLIDE: ĐỐI TƯỢNG VÀ PHẠM VI NGHIÊN CỨU
% ======================================================
\begin{frame}
    \frametitle{Đối tượng và phạm vi nghiên cứu}

    \begin{BulletTight}
        \item \textbf{Đối tượng nghiên cứu:}
        \begin{itemize}
            \setlength{\itemsep}{0.3em}
            \item Hệ thống đa tác tử (Multi-Agent System) tích hợp mô hình ngôn ngữ lớn (LLM) cho kiểm thử phần mềm tự động.
            \item Các tác tử chính: Planner, UI, API, Evaluator, Coordinator, HITL Agent\ldots
            \item Cơ chế phối hợp, giao tiếp và phản hồi giữa các tác tử trong quá trình kiểm thử.
        \end{itemize}

        \vspace{0.7em}
        \item \textbf{Phạm vi nghiên cứu:}
        \begin{itemize}
            \setlength{\itemsep}{0.3em}
            \item Ứng dụng cho kiểm thử phần mềm web và di động.
            \item Tập trung phát triển và phân tích sâu \textbf{một tác tử tiêu biểu} nhằm tối ưu hiệu năng và khả năng phối hợp.

        \end{itemize}


    \end{BulletTight}
\end{frame}
% ======================================================
% SLIDE: PHƯƠNG PHÁP NGHIÊN CỨU – THIẾT KẾ THỰC NGHIỆM
% ======================================================
\begin{frame}
    \frametitle{Phương pháp nghiên cứu}

    \begin{BulletTight}
        \item \textbf{1. Khảo sát và phân tích:}
        Tổng hợp các công trình trong và ngoài nước, xác định khoảng trống và định hướng nghiên cứu.

        \item \textbf{2. Thiết kế mô hình tổng thể:}
        Xây dựng khung hệ thống đa tác tử tích hợp LLM; xác định vai trò, giao tiếp và cơ chế phối hợp.

        \item \textbf{3. Phát triển nguyên mẫu:}
        Triển khai các tác tử chính (Planner, UI, API, Coordinator \ldots) và thiết lập cơ chế HITL.

        \item \textbf{4. Thực nghiệm và đánh giá:}
        Kiểm thử thực tế, đo lường hiệu năng, độ chính xác, độ tin cậy và chi phí.
        So sánh với baseline (hệ LLM đơn tác tử - đa tác tử) để kiểm chứng hiệu quả.

        \item \textbf{5. Hoàn thiện và tổng kết:}
        Phân tích kết quả, rút ra tiêu chí tối ưu và đề xuất mô hình cải tiến.


    \end{BulletTight}
\end{frame}
