\begin{frame}
    \frametitle{Ý nghĩa}

    \begin{BulletTight}
        \item \textbf{Ý nghĩa khoa học:}
        \begin{itemize}
            \setlength{\itemsep}{0.15em}
            \item Mở rộng nghiên cứu kiểm thử tự động dựa trên LLM và hệ đa tác tử.
            \item Đề xuất cơ chế phối hợp tác tử để tăng độ chính xác và ổn định.
            \item Làm rõ vai trò HITL trong tăng độ tin cậy và giảm ảo giác.
            \item Đưa ra framework có khả năng tái sử dụng và mở rộng.
        \end{itemize}

        \vspace{0.4em}

        \item \textbf{Ý nghĩa thực tiễn:}
        \begin{itemize}
            \setlength{\itemsep}{0.15em}
            \item Công cụ kiểm thử tự động tích hợp DevOps/CI-CD.
            \item Giảm chi phí, công sức kiểm thử thủ công.
            \item Tăng bao phủ kiểm thử, chất lượng và tốc độ phát hành phần mềm.
            \item Thúc đẩy ứng dụng AI trong phát triển phần mềm thực tế.
        \end{itemize}
    \end{BulletTight}
\end{frame}
