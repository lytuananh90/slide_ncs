\begin{frame}
    \frametitle{Ý nghĩa khoa học và thực tiễn}

    \begin{BulletTight}
        \item \textbf{Ý nghĩa khoa học:}
        Đóng góp tri thức mới về kiểm thử tự động dựa trên LLM và hệ đa tác tử;
        đề xuất cơ chế phối hợp tăng độ chính xác, hiệu quả kiểm thử;
        làm rõ vai trò HITL trong giảm ảo giác;
        xây dựng framework có khả năng tái sử dụng.

        \vspace{0.6em}
        \item \textbf{Ý nghĩa thực tiễn:}
        Phát triển công cụ kiểm thử tự động tích hợp DevOps/CI-CD;
        giảm chi phí, công sức thủ công;
        tăng bao phủ, chất lượng và tốc độ kiểm thử;
        thúc đẩy ứng dụng AI trong công nghiệp phần mềm.

        \vspace{0.5em}
        \item[\raise0.2ex\hbox{\color{KHTN_BLUE!85}\scriptsize\faArrowRight}]
        \textbf{Tổng kết:} Hệ kiểm thử thông minh, tin cậy, mở rộng — kết hợp giá trị khoa học và ứng dụng.
    \end{BulletTight}
\end{frame}
